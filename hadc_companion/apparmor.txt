#include <tunables/global>

profile hadc_companion flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  signal,
  network inet stream,
  network inet6 stream,

  # S6 Overlay startup/runtime. We intentionally do NOT use the broad
  # AppArmor "file," rule from the generic example profile, so read
  # permissions for shell scripts must be explicit here.
  /init rix,
  /bin/** rix,
  /usr/bin/** rix,
  /usr/local/bin/** rix,
  /run/{s6,s6-rc*,service}/** rix,
  /package/** rix,
  /command/** rix,
  /etc/s6-overlay/** rwix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,

  # Python runtime and shared libraries.
  /usr/lib/** mr,
  /usr/local/lib/** mr,
  /lib/** mr,
  /etc/ssl/** r,
  /etc/passwd r,
  /etc/group r,
  /proc/sys/crypto/fips_enabled r,
  /dev/null rw,
  /dev/urandom r,
  /tmp/** rwk,

  # Companion application code is immutable at runtime.
  /opt/hadc-companion/ r,
  /opt/hadc-companion/** r,

  # Companion persistent lifecycle state.
  /data/ rw,
  /data/tls/ rwk,
  /data/**/ rwk,
  /data/** rwk,

  # Home Assistant configuration stays read-only at both the Docker
  # mount and AppArmor layers. Internal storage remains denied.
  /homeassistant/ r,
  /homeassistant/** r,
  deny /homeassistant/.storage/ r,
  deny /homeassistant/.storage/** r,
}
